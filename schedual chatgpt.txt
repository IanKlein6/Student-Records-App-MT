


### dbdiagram.io ###

Table students {
  id int [pk, increment]
  first_name varchar
  last_name varchar
  email varchar
  active bool
}

Table groups {
  id int [pk, increment]
  name varchar
  scheduled_date date
  semester_id int [ref: > semesters.id]
}

Table student_group_memberships {
  id int [pk, increment]
  student_id int [ref: > students.id]
  group_id int [ref: > groups.id]
}

Table semesters {
  id int [pk, increment]
  name varchar
  start_date date
  end_date date
}

Table trials {
  id int [pk, increment]
  student_id int [ref: > students.id]
  group_id int [ref: > groups.id]
  type varchar  // 'question' or 'protocol'
  attempt int
  result varchar  // 'pass', 'fail'
  comment text
  created_at datetime
}

Table technika {
  id int [pk, increment]
  name varchar
  email varchar
}

Table availability_slots {
  id int [pk, increment]
  technika_id int [ref: > technika.id]
  start_time datetime
  end_time datetime
  status varchar  // 'available', 'unavailable', 'booked'
}

Table appointments {
  id int [pk, increment]
  student_id int [ref: > students.id]
  technika_id int [ref: > technika.id]
  slot_id int [ref: > availability_slots.id]
  trial_type varchar
  calendar_event_id varchar
}
